cmake_minimum_required(VERSION 3.22)
project(SwellDelay VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(DELAY_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND DELAY_FORMATS AU)
endif()
# Never change PLUGIN_CODE / BUNDLE_ID once released: saved DAW sessions would lose the plugin.
juce_add_plugin(SwellDelay COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.delay" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Sdly FORMATS ${DELAY_FORMATS} PRODUCT_NAME "Swell Delay" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(DelayAssets SOURCES swell-mark.png)
target_sources(SwellDelay PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SwellDelay PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SwellDelay PRIVATE DelayAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(DelayTest PRODUCT_NAME "Delay Audio Tests")
target_sources(DelayTest PRIVATE Tests/AudioTest.cpp)
target_include_directories(DelayTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(DelayTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(DelayTest PRIVATE SwellDelay)
add_test(NAME DelayAudio COMMAND DelayTest)
juce_add_console_app(DelaySnapshot PRODUCT_NAME "Delay Snapshot")
target_sources(DelaySnapshot PRIVATE Tools/Snapshot.cpp)
target_include_directories(DelaySnapshot PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(DelaySnapshot PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(DelaySnapshot PRIVATE SwellDelay)
juce_add_console_app(DelayDemo PRODUCT_NAME "Delay Demo Renderer")
target_sources(DelayDemo PRIVATE Tools/RenderDemo.cpp)
target_include_directories(DelayDemo PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(DelayDemo PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(DelayDemo PRIVATE SwellDelay)
